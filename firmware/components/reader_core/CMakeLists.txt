if(ESP_PLATFORM)
    idf_component_register(SRCS "core.cpp" "diagnostics.cpp" INCLUDE_DIRS "include" PRIV_REQUIRES esp_timer freertos)
    target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
else()
    add_library(reader_core core.cpp diagnostics.cpp)
    target_include_directories(reader_core PUBLIC include)
    target_compile_features(reader_core PUBLIC cxx_std_17)
    target_compile_options(reader_core PRIVATE -Wall -Wextra -Werror)
    option(PICO_PAGE_HOST_PERF_METRICS "Exercise operation metrics in shared host code" OFF)
    option(PICO_PAGE_HOST_TRACE_EVENTS "Exercise bounded events in shared host code" OFF)
    if(PICO_PAGE_HOST_PERF_METRICS OR PICO_PAGE_HOST_TRACE_EVENTS)
        find_package(Threads REQUIRED)
        target_link_libraries(reader_core PUBLIC Threads::Threads)
        target_compile_definitions(reader_core PUBLIC CONFIG_PICO_PAGE_PERF_METRICS=1)
    endif()
    if(PICO_PAGE_HOST_TRACE_EVENTS)
        target_compile_definitions(reader_core PUBLIC CONFIG_PICO_PAGE_TRACE_EVENTS=1)
    endif()
endif()
